        // State variables
        let watchlist = JSON.parse(localStorage.getItem('anime_watchlist')) || [1];

        // Initialize application on load
        window.onload = function() {
            renderTrending();
            renderCatalog();
            updateWatchlistUI();
        };

        // Navigation Tab Switcher
        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('text-red-500');
                btn.classList.add('text-gray-400');
                if(btn.dataset.tab === tabId) {
                    btn.classList.add('text-red-500');
                    btn.classList.remove('text-gray-400');
                }
            });
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Toggle Mobile Menu
        function toggleMobileMenu() {
            const menu = document.getElementById('mobile-menu');
            menu.classList.toggle('hidden');
        }

        // Render Trending Anime (Top 4)
        function renderTrending() {
            const grid = document.getElementById('trending-grid');
            const trending = [...animeData].sort((a,b) => b.rating - a.rating).slice(0, 4);
            grid.innerHTML = trending.map(anime => createAnimeCard(anime)).join('');
        }

        // Render Full Catalog
        function renderCatalog(filteredList = animeData) {
            const grid = document.getElementById('catalog-grid');
            if(filteredList.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-16 text-gray-500">No anime matches your criteria.</div>`;
                return;
            }
            grid.innerHTML = filteredList.map(anime => createAnimeCard(anime)).join('');
        }

        // Helper: Create Anime Card HTML
        function createAnimeCard(anime) {
            const isSaved = watchlist.includes(anime.id);
            return `
                <div class="glass-panel rounded-3xl overflow-hidden card-hover border border-gray-800 flex flex-col justify-between group">
                    <div class="relative overflow-hidden aspect-[4/5] cursor-pointer" onclick="openAnimeModal(${anime.id})">
                        <img src="${anime.image}" alt="${anime.title}" class="w-full h-full object-cover group-hover:scale-110 transition duration-500">
                        <div class="absolute inset-0 bg-gradient-to-t from-darkBg via-transparent to-transparent opacity-80"></div>
                        <span class="absolute top-3 left-3 bg-red-600/90 backdrop-blur-md text-white text-xs font-bold px-3 py-1 rounded-full shadow-lg">
                            ${anime.genre}
                        </span>
                        <button onclick="event.stopPropagation(); toggleWatchlist(${anime.id})" class="absolute top-3 right-3 w-9 h-9 rounded-full glass-panel flex items-center justify-center text-white hover:bg-red-600 transition shadow-lg">
                            <i class="${isSaved ? 'fa-solid fa-bookmark text-red-500' : 'fa-regular fa-bookmark'}"></i>
                        </button>
                        <div class="absolute bottom-3 left-3 right-3 flex items-center justify-between text-xs text-gray-300 font-medium">
                            <span><i class="fa-solid fa-star text-amber-400 mr-1"></i> ${anime.rating}</span>
                            <span>${anime.episodes} Eps • ${anime.year}</span>
                        </div>
                    </div>
                    <div class="p-5 space-y-3">
                        <h3 onclick="openAnimeModal(${anime.id})" class="font-bold text-lg text-white hover:text-red-500 cursor-pointer transition line-clamp-1">${anime.title}</h3>
                        <p class="text-xs text-gray-400 line-clamp-2">${anime.synopsis}</p>
                        <button onclick="openAnimeModal(${anime.id})" class="w-full bg-gray-900/80 hover:bg-red-600 text-gray-300 hover:text-white py-2 rounded-xl text-sm font-semibold transition flex items-center justify-center space-x-2">
                            <i class="fa-solid fa-circle-info"></i>
                            <span>Details & Watch</span>
                        </button>
                    </div>
                </div>
            `;
        }

        // Apply Genre and Sorting Filters
        function applyFilters() {
            const genre = document.getElementById('genre-select').value;
            const sort = document.getElementById('sort-select').value;

            let list = [...animeData];
            if(genre !== 'All') {
                list = list.filter(a => a.genre === genre);
            }

            if(sort === 'rating') {
                list.sort((a,b) => b.rating - a.rating);
            } else if(sort === 'popularity') {
                list.sort((a,b) => b.popularity - a.popularity);
            } else if(sort === 'newest') {
                list.sort((a,b) => b.year - a.year);
            }

            renderCatalog(list);
        }

        // Filter by Vibe shortcut
        function filterByGenre(genreName) {
            switchTab('explore');
            document.getElementById('genre-select').value = genreName;
            applyFilters();
        }

        // Global Search Handler
        function handleGlobalSearch(query) {
            const q = query.toLowerCase().trim();
            if(!q) {
                renderCatalog(animeData);
                return;
            }
            const filtered = animeData.filter(a => a.title.toLowerCase().includes(q) || a.genre.toLowerCase().includes(q) || a.synopsis.toLowerCase().includes(q));
            switchTab('explore');
            renderCatalog(filtered);
        }

        // Surprise Me / Random Anime
        function openRandomAnime() {
            const random = animeData[Math.floor(Math.random() * animeData.length)];
            openAnimeModal(random.id);
        }

        // Watchlist Management
        function toggleWatchlist(animeId) {
            const index = watchlist.indexOf(animeId);
            if(index > -1) {
                watchlist.splice(index, 1);
                showToast("Removed from Watchlist", "info");
            } else {
                watchlist.push(animeId);
                showToast("Added to Watchlist successfully!", "success");
            }
            localStorage.setItem('anime_watchlist', JSON.stringify(watchlist));
            updateWatchlistUI();
            renderTrending();
            renderCatalog();
            // Update modal if open
            const modal = document.getElementById('anime-modal');
            if(!modal.classList.contains('hidden')) {
                // Refresh modal state
            }
        }

        function updateWatchlistUI() {
            document.getElementById('watchlist-badge').textContent = watchlist.length;
            document.getElementById('watchlist-count').textContent = watchlist.length;

            const grid = document.getElementById('watchlist-grid');
            const emptyState = document.getElementById('watchlist-empty');

            const savedAnime = animeData.filter(a => watchlist.includes(a.id));
            if(savedAnime.length === 0) {
                grid.innerHTML = '';
                emptyState.classList.remove('hidden');
            } else {
                emptyState.classList.add('hidden');
                grid.innerHTML = savedAnime.map(anime => createAnimeCard(anime)).join('');
            }
        }

        // Modal Preview Functions
        function openAnimeModal(animeId) {
            const anime = animeData.find(a => a.id === animeId);
            if(!anime) return;

            const isSaved = watchlist.includes(anime.id);
            const modalContent = document.getElementById('modal-content');
            modalContent.innerHTML = `
                <div class="relative h-72 sm:h-96 w-full">
                    <img src="${anime.image}" alt="${anime.title}" class="w-full h-full object-cover">
                    <div class="absolute inset-0 bg-gradient-to-t from-[#090a0f] via-[#090a0f]/50 to-transparent"></div>
                    <div class="absolute bottom-6 left-6 right-6 space-y-2">
                        <div class="flex items-center space-x-2">
                            <span class="bg-red-600 text-white text-xs font-bold px-3 py-1 rounded-full">${anime.genre}</span>
                            <span class="bg-gray-800 text-gray-300 text-xs font-medium px-3 py-1 rounded-full">${anime.status}</span>
                        </div>
                        <h2 class="text-3xl sm:text-5xl font-black text-white">${anime.title}</h2>
                    </div>
                </div>
                <div class="p-6 sm:p-8 space-y-6">
                    <div class="flex flex-wrap items-center justify-between gap-4 border-b border-gray-800 pb-6">
                        <div class="flex items-center space-x-6 text-sm text-gray-300">
                            <div><i class="fa-solid fa-star text-amber-400 mr-1.5"></i> <strong class="text-white">${anime.rating}</strong> / 10</div>
                            <div><i class="fa-solid fa-film text-red-500 mr-1.5"></i> ${anime.episodes} Episodes</div>
                            <div><i class="fa-solid fa-calendar text-blue-400 mr-1.5"></i> ${anime.year}</div>
                        </div>
                        <div class="flex items-center space-x-3">
                            <button onclick="toggleWatchlist(${anime.id})" class="glass-panel hover:bg-gray-800 text-white font-semibold px-5 py-2.5 rounded-xl flex items-center space-x-2 transition border border-gray-700">
                                <i class="${isSaved ? 'fa-solid fa-bookmark text-red-500' : 'fa-regular fa-bookmark'}"></i>
                                <span>${isSaved ? 'In Watchlist' : 'Add to Watchlist'}</span>
                            </button>
                            <button onclick="playAnimeStream('${anime.title}')" class="bg-red-600 hover:bg-red-700 text-white font-bold px-6 py-2.5 rounded-xl shadow-lg shadow-red-600/30 flex items-center space-x-2 transition">
                                <i class="fa-solid fa-play"></i>
                                <span>Stream Now</span>
                            </button>
                        </div>
                    </div>
                    <div class="space-y-3">
                        <h3 class="font-bold text-lg text-white">Synopsis</h3>
                        <p class="text-gray-300 text-sm sm:text-base leading-relaxed">${anime.synopsis}</p>
                    </div>
                    <div class="space-y-3">
                        <h3 class="font-bold text-lg text-white">Featured Characters / Cast</h3>
                        <div class="flex flex-wrap gap-2">
                            ${anime.cast.map(c => `<span class="bg-gray-900 border border-gray-800 px-3 py-1.5 rounded-xl text-xs text-gray-300 font-medium">${c}</span>`).join('')}
                        </div>
                    </div>
                </div>
            `;

            const modal = document.getElementById('anime-modal');
            const container = document.getElementById('modal-container');
            modal.classList.remove('hidden');
            setTimeout(() => {
                modal.classList.remove('opacity-0');
                container.classList.remove('scale-95');
            }, 10);
        }

        function closeAnimeModal() {
            const modal = document.getElementById('anime-modal');
            const container = document.getElementById('modal-container');
            modal.classList.add('opacity-0');
            container.classList.add('scale-95');
            setTimeout(() => {
                modal.classList.add('hidden');
            }, 300);
        }

        // Simulated Streaming player notification
        function playAnimeStream(title) {
            showToast(`Loading episode stream for "${title}"... Enjoy!`, 'success');
            closeAnimeModal();
        }

        // Toast Notification System
        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = `glass-panel px-5 py-3.5 rounded-2xl shadow-2xl flex items-center space-x-3 text-sm font-medium text-white pointer-events-auto transform translate-y-4 opacity-0 transition-all duration-300 border ${type === 'success' ? 'border-red-500/50 bg-red-950/40' : 'border-gray-700'}`;
            toast.innerHTML = `
                <i class="fa-solid ${type === 'success' ? 'fa-circle-check text-red-500' : 'fa-circle-info text-blue-400'} text-lg"></i>
                <span>${message}</span>
            `;
            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.remove('translate-y-4', 'opacity-0');
            }, 50);

            setTimeout(() => {
                toast.classList.add('translate-y-4', 'opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }

        // Close modal on background click
        document.getElementById('anime-modal').addEventListener('click', function(e) {
            if(e.target === this) closeAnimeModal();
        });
    </script>
